cmake_minimum_required(VERSION 3.16)
project(kart_system)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
ament_auto_add_library(kart_system_components SHARED src/operation_nodes.cpp src/joy_manager_node.cpp)
target_compile_features(kart_system_components PUBLIC cxx_std_17)
target_compile_options(kart_system_components PRIVATE -Wall -Wextra -Wpedantic)
rclcpp_components_register_node(kart_system_components PLUGIN "kart_system::ModeManagerNode" EXECUTABLE kart_operation_mode_manager)
rclcpp_components_register_node(kart_system_components PLUGIN "kart_system::CommandMuxNode" EXECUTABLE kart_command_mux)
rclcpp_components_register_node(kart_system_components PLUGIN "kart_system::JoyManagerNode" EXECUTABLE kart_joy_manager_node)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_control_core test/test_control_core.cpp)
  target_compile_features(test_control_core PRIVATE cxx_std_17)
  target_include_directories(test_control_core PRIVATE include)
  target_compile_options(test_control_core PRIVATE -UNDEBUG)
  add_test(NAME control_core COMMAND test_control_core)
  add_executable(test_joy_controls test/test_joy_controls.cpp)
  target_compile_features(test_joy_controls PRIVATE cxx_std_17)
  target_include_directories(test_joy_controls PRIVATE include)
  target_compile_options(test_joy_controls PRIVATE -UNDEBUG)
  add_test(NAME joy_controls COMMAND test_joy_controls)
endif()
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_auto_package(USE_SCOPED_HEADER_INSTALL_DIR INSTALL_TO_SHARE config)
