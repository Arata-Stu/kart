cmake_minimum_required(VERSION 3.16)
project(kart_parking)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/parking_planner scripts/parking_plan DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(adapter test/test_adapter.py)
  ament_add_pytest_test(tracking test/test_tracking.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}:${CMAKE_CURRENT_SOURCE_DIR}/../kart_hdmap)
  ament_add_pytest_test(parking test/test_parking.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}:${CMAKE_CURRENT_SOURCE_DIR}/../kart_hdmap)
endif()
ament_auto_package(INSTALL_TO_SHARE config)
