cmake_minimum_required(VERSION 3.16)
project(kart_e2e)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(ament_cmake_python REQUIRED)
find_package(CUDAToolkit REQUIRED)
ament_auto_add_library(kart_e2e_components SHARED src/control_decoder_node.cpp)
target_compile_features(kart_e2e_components PUBLIC cxx_std_17)
target_link_libraries(kart_e2e_components CUDA::cudart)
rclcpp_components_register_node(kart_e2e_components PLUGIN "kart_e2e::ControlDecoderNode" EXECUTABLE e2e_control_decoder EXECUTOR MultiThreadedExecutor)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/e2e_evaluate scripts/e2e_export_onnx scripts/e2e_train scripts/e2e_preprocess_bag DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
include(CTest)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(contract test/test_contract.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  ament_add_pytest_test(evaluate test/test_evaluate.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  ament_add_pytest_test(export test/test_export.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  ament_add_pytest_test(model test/test_model.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(test_control_gate test/test_control_gate.cpp)
  target_compile_features(test_control_gate PRIVATE cxx_std_17)
  target_include_directories(test_control_gate PRIVATE include)
  target_compile_options(test_control_gate PRIVATE -UNDEBUG)
  add_test(NAME control_gate COMMAND test_control_gate)
  ament_add_pytest_test(data test/test_data.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
endif()
ament_auto_package(USE_SCOPED_HEADER_INSTALL_DIR INSTALL_TO_SHARE config)
