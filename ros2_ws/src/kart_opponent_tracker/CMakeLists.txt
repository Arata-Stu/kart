cmake_minimum_required(VERSION 3.16)
project(kart_opponent_tracker)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(Eigen3 REQUIRED)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ByteTrack-cpp/CMakeLists.txt")
  message(FATAL_ERROR "ByteTrack-cpp is missing. Run bash scripts/repos.sh import from kart first.")
endif()
add_subdirectory(vendor/ByteTrack-cpp EXCLUDE_FROM_ALL)
ament_auto_add_library(opponent_tracker_component SHARED src/opponent_tracker.cpp)
target_compile_features(opponent_tracker_component PUBLIC cxx_std_17)
target_link_libraries(opponent_tracker_component bytetrack)
rclcpp_components_register_node(opponent_tracker_component
  PLUGIN "kart_opponent_tracker::OpponentTracker" EXECUTABLE opponent_tracker)
install(TARGETS bytetrack LIBRARY DESTINATION lib)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/opponent_ground DESTINATION lib/${PROJECT_NAME})
install(FILES README.md vendor/ByteTrack-cpp/LICENSE DESTINATION share/${PROJECT_NAME})
ament_auto_package(INSTALL_TO_SHARE config)
