cmake_minimum_required(VERSION 3.16)
project(kart_opponent_tracker)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(Eigen3 REQUIRED)
find_path(BYTETRACK_INCLUDE_DIR NAMES ByteTrack/BYTETracker.h)
find_library(BYTETRACK_LIBRARY NAMES bytetrack)
if(NOT BYTETRACK_INCLUDE_DIR OR NOT BYTETRACK_LIBRARY)
  message(FATAL_ERROR "ByteTrack-cpp development files are missing. Rebuild the kart Docker image.")
endif()
ament_auto_add_library(opponent_tracker_component SHARED src/opponent_tracker.cpp)
target_compile_features(opponent_tracker_component PUBLIC cxx_std_17)
target_include_directories(opponent_tracker_component PRIVATE "${BYTETRACK_INCLUDE_DIR}")
target_link_libraries(opponent_tracker_component "${BYTETRACK_LIBRARY}" Eigen3::Eigen)
rclcpp_components_register_node(opponent_tracker_component
  PLUGIN "kart_opponent_tracker::OpponentTracker" EXECUTABLE opponent_tracker)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/opponent_ground scripts/opponent_debug_image DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(debug_image test/test_debug_image.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  ament_add_pytest_test(ground_status test/test_ground_status.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  add_executable(test_tracked_class test/test_tracked_class.cpp)
  target_compile_features(test_tracked_class PRIVATE cxx_std_17)
  target_include_directories(test_tracked_class PRIVATE src)
  add_test(NAME tracked_class COMMAND test_tracked_class)
endif()
ament_auto_package(INSTALL_TO_SHARE config)
