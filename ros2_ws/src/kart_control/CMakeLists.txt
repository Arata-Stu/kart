cmake_minimum_required(VERSION 3.16)
project(kart_control)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
find_package(ackermann_msgs REQUIRED)
ament_auto_add_library(kart_control_components SHARED src/pure_pursuit_node.cpp src/speed_controller_node.cpp)
ament_target_dependencies(kart_control_components ackermann_msgs)
target_compile_features(kart_control_components PUBLIC cxx_std_17)
target_compile_options(kart_control_components PRIVATE -Wall -Wextra -Wpedantic)
rclcpp_components_register_node(kart_control_components PLUGIN "kart_control::PurePursuitNode" EXECUTABLE pure_pursuit)
rclcpp_components_register_node(kart_control_components PLUGIN "kart_control::SpeedControllerNode" EXECUTABLE speed_controller)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_speed_pid test/test_speed_pid.cpp)
  target_compile_features(test_speed_pid PRIVATE cxx_std_17)
  target_include_directories(test_speed_pid PRIVATE include)
  target_compile_options(test_speed_pid PRIVATE -UNDEBUG)
  add_test(NAME speed_pid COMMAND test_speed_pid)
  add_executable(test_pure_pursuit test/test_pure_pursuit.cpp)
  target_compile_features(test_pure_pursuit PRIVATE cxx_std_17)
  target_include_directories(test_pure_pursuit PRIVATE include)
  target_compile_options(test_pure_pursuit PRIVATE -UNDEBUG)
  add_test(NAME pure_pursuit COMMAND test_pure_pursuit)
endif()
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_auto_package(USE_SCOPED_HEADER_INSTALL_DIR INSTALL_TO_SHARE config)
