cmake_minimum_required(VERSION 3.16)
project(kart_vehicle)
find_package(ament_cmake_auto REQUIRED)
ament_auto_find_build_dependencies()
ament_auto_add_library(kart_vehicle_component SHARED src/bridge_protocol.cpp src/bridge_node.cpp)
target_compile_features(kart_vehicle_component PUBLIC cxx_std_17)
target_compile_options(kart_vehicle_component PRIVATE -Wall -Wextra -Wpedantic)
rclcpp_components_register_node(kart_vehicle_component PLUGIN "kart_vehicle::BridgeNode" EXECUTABLE kart_bridge_node)
include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(test_bridge_core test/test_bridge_core.cpp src/bridge_protocol.cpp)
  target_compile_features(test_bridge_core PRIVATE cxx_std_17)
  target_include_directories(test_bridge_core PRIVATE include ${kart_system_INCLUDE_DIRS})
  target_link_libraries(test_bridge_core PRIVATE Threads::Threads)
  target_compile_options(test_bridge_core PRIVATE -UNDEBUG)
  add_test(NAME bridge_core COMMAND test_bridge_core)
  add_executable(test_esc_state test/test_esc_state.cpp)
  target_compile_features(test_esc_state PRIVATE cxx_std_17)
  target_include_directories(test_esc_state PRIVATE include ${kart_system_INCLUDE_DIRS})
  target_compile_options(test_esc_state PRIVATE -UNDEBUG)
  add_test(NAME esc_state COMMAND test_esc_state)
endif()
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_auto_package(USE_SCOPED_HEADER_INSTALL_DIR)
